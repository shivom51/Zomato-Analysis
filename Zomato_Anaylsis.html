import seaborn as sns
import matplotlib.pyplot as plt

# create a pivot table with restaurant type vs online order counts
pivot_table = df1.pivot_table(
    index='listed_in(type)',      # rows → restaurant types
    columns='online_order',       # columns → Yes/No
    aggfunc='size',               # count number of entries
    fill_value=0                  # replace missing values with 0
)

# plot heatmap
plt.figure(figsize=(6,4))
sns.heatmap(pivot_table, annot=True, cmap="YlGnBu", fmt='d')  # annot=True to show numbers
plt.title("Heatmap")
plt.xlabel("Online Order")
plt.ylabel("Listed In (Type)")
plt.tight_layout()
plt.show()
